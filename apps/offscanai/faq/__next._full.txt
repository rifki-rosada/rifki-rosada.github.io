                const links = document.querySelectorAll('a[href]');
                const options = { rootMargin: "300px" };

                const preload = (url) => {
                  if (!url || url.startsWith("#") || url.startsWith("mailto:")) return;
                  const link = document.createElement("link");
                  link.rel = "prefetch";
                  link.href = url;
                  document.head.appendChild(link);
                };

                // ✅ Preload when link scrolls near viewport
                const observer = new IntersectionObserver((entries) => {
                  entries.forEach(entry => {
                    if (entry.isIntersecting) {
                      preload(entry.target.getAttribute('href'));
                      observer.unobserve(entry.target);
                    }
                  });
                }, options);

                links.forEach(link => observer.observe(link));

                // ✅ Ultra Mode: Preload on Hover
                links.forEach(link => {
                  link.addEventListener('mouseenter', () => {
                    preload(link.getAttribute('href'));
                  });
                });

                // ✅ Ultra Mode: Preload on Touch (mobile)
                links.forEach(link => {
                  link.addEventListener('touchstart', () => {
                    preload(link.getAttribute('href'));
                  }, { passive: true });
                });
